#include <algorithm>
#include <array>
#include <cmath>
#include <fstream>
#include <iostream>
#include <utility>
#include <vector>

#include "config.h"
#include "gait_planner.h"
#include "acados_solver_lipm_opti_pessi.h"
#include "acados_c/ocp_nlp_interface.h"

namespace {

constexpr int NX = 22;
constexpr int NU = 22;
constexpr int NP = 8;
constexpr int NX_SINGLE = 11;

using State = std::array<double, NX>;
using Control = std::array<double, NU>;
using RobotState = std::array<double, NX_SINGLE>;

[[nodiscard]] double norm2D(const std::array<double, 2>& v)
{
    return std::hypot(v[0], v[1]);
}

[[nodiscard]] double dot2D(const std::array<double, 2>& a,
                           const std::array<double, 2>& b)
{
    return a[0] * b[0] + a[1] * b[1];
}

[[nodiscard]] std::array<double, 2> closestPoint(
    const std::array<double, 2>& com,
    const std::array<double, 4>& hips,
    const std::array<double, 2>& obstacle)
{
    const std::array<double, 2> hip0{hips[0], hips[1]};
    const std::array<double, 2> hip1{hips[2], hips[3]};

    const double d_com = norm2D({obstacle[0] - com[0], obstacle[1] - com[1]});
    const double d_hip0 = norm2D({obstacle[0] - hip0[0], obstacle[1] - hip0[1]});
    const double d_hip1 = norm2D({obstacle[0] - hip1[0], obstacle[1] - hip1[1]});

    if (d_hip0 <= d_com && d_hip0 <= d_hip1) return hip0;
    if (d_hip1 <= d_com && d_hip1 <= d_hip0) return hip1;
    return com;
}

[[nodiscard]] std::pair<std::array<double, 2>, double> separatingLine(
    const std::array<double, 2>& closest,
    const std::array<double, 2>& obstacle,
    double obstacle_margin)
{
    // Il modello impone a·p+b <= 0 per robot/feet e
    // a·obs+b >= 1.8*r_obstacle. Posizionare la retta al punto medio
    // non garantisce quest'ultimo vincolo quando la separazione è ridotta.
    const std::array<double, 2> dir{
        obstacle[0] - closest[0], obstacle[1] - closest[1]};
    const double dist = norm2D(dir);
    if (dist < 1e-8) {
        // Normale di fallback; in questa situazione la fattibilità va
        // comunque verificata, perché robot e ostacolo quasi coincidono.
        const std::array<double, 2> a{1.0, 0.0};
        return {a, -obstacle[0] + obstacle_margin};
    }
    const std::array<double, 2> a{dir[0] / dist, dir[1] / dist};
    // In questo modo il vincolo dell'ostacolo è soddisfatto al limite;
    // i punti del robot restano fattibili solo se hanno abbastanza distanza.
    const double b = -dot2D(a, obstacle) + obstacle_margin;
    return {a, b};
}

void fillControlHalf(double* dst,
                     const std::array<double, 4>& hips,
                     double dt,
                     const std::pair<std::array<double, 2>, double>& line)
{
    dst[0] = hips[0];
    dst[1] = hips[1];
    dst[2] = hips[2];
    dst[3] = hips[3];
    dst[4] = 0.5;       // alpha
    dst[5] = 0.0;       // beta
    dst[6] = 0.0;       // gamma
    dst[7] = dt;
    dst[8] = line.first[0];
    dst[9] = line.first[1];
    dst[10] = line.second;
}

template <class Gait>
void setStageParameters(lipm_opti_pessi_solver_capsule* capsule,
                        int stage,
                        const Gait& gait,
                        const GaitPlanner& planner,
                        const std::array<double, 2>& obstacle)
{
    const auto offset0 = planner.get_hip_offset(gait.first);
    const auto offset1 = planner.get_hip_offset(gait.second);
    // Come nel Python: il raggio passato è già r_obs + 0.1;
    // il modello aggiunge poi y_dot_max * x_pe[10] nel ramo pessimistico.
    const double r_dyn = Config::r_obs + 0.1;
    double p[NP] = {
        offset0[0], offset0[1], offset1[0], offset1[1],
        obstacle[0], obstacle[1], r_dyn, Config::y_dot_max
    };
    lipm_opti_pessi_acados_update_params(capsule, stage, p, NP);
}

} // namespace

int main()
{
    std::cout << "--- Starting Simulation MPC in C++ ---\n";

    constexpr int N = Config::N_horizon;
    if (N < 1) {
        std::cerr << "N_horizon deve essere almeno 1.\n";
        return 1;
    }

    auto* capsule = lipm_opti_pessi_acados_create_capsule();
    if (capsule == nullptr) {
        std::cerr << "Impossibile creare la capsule Acados.\n";
        return 1;
    }
    if (lipm_opti_pessi_acados_create(capsule) != 0) {
        std::cerr << "Errore durante l'inizializzazione del solver Acados.\n";
        lipm_opti_pessi_acados_free_capsule(capsule);
        return 1;
    }

    GaitPlanner gait_planner(Config::off_x, Config::off_y, Config::sim_per_phase);

    RobotState X_sim{};
    X_sim[0] = Config::x_init_x;
    X_sim[1] = Config::x_init_y;
    X_sim[2] = Config::theta_init;

    const auto initial_gait_horizon = gait_planner.get_gait_horizon(0, N + 1);
    if (initial_gait_horizon.size() < static_cast<std::size_t>(N + 1)) {
        std::cerr << "get_gait_horizon deve restituire almeno N_horizon + 1 elementi.\n";
        lipm_opti_pessi_acados_free(capsule);
        lipm_opti_pessi_acados_free_capsule(capsule);
        return 1;
    }
    const auto initial_hips = gait_planner.compute_hip_positions(
        {X_sim[0], X_sim[1], X_sim[2]}, initial_gait_horizon[0]);
    X_sim[6] = initial_hips[0];
    X_sim[7] = initial_hips[1];
    X_sim[8] = initial_hips[2];
    X_sim[9] = initial_hips[3];

    State X_aug{};
    std::copy(X_sim.begin(), X_sim.end(), X_aug.begin());
    std::copy(X_sim.begin(), X_sim.end(), X_aug.begin() + NX_SINGLE);

    // Warm start iniziale: stesso stato duplicato in tutti i nodi, come nel Python.
    std::vector<State> X_sol_t_prec(static_cast<std::size_t>(N + 1), X_aug);
    Control u_init{};
    // Nel modello i limiti su dt sono divisi per steps_per_phase.
    // Anche il guess iniziale deve essere nello stesso intervallo.
    const double dt_guess = Config::dt_max / Config::sim_per_phase;
    u_init[7] = dt_guess;
    u_init[18] = dt_guess;
    std::vector<Control> U_sol_t_prec(static_cast<std::size_t>(N), u_init);

    std::ofstream log_file("mpc_log.csv");
    if (!log_file) {
        std::cerr << "Impossibile aprire mpc_log.csv in scrittura.\n";
        lipm_opti_pessi_acados_free(capsule);
        lipm_opti_pessi_acados_free_capsule(capsule);
        return 1;
    }
    log_file << "step,x,y,theta,vx,vy\n";

    double t_global = 0.0;
    int reached_target_flag = 0;
    double dt_chosen = Config::dt_max;

    for (int step = 0; step < Config::sim_steps; ++step) {
        // In questo esempio l'ostacolo è statico, come nel ramo static del Python.
        const std::array<double, 2> obstacle{
            Config::obs_pos_init[0], Config::obs_pos_init[1]};
        (void)t_global; // Usato quando si implementano qui i tipi dinamici di ostacolo.

        X_sim[10] = 0.0;
        std::copy(X_sim.begin(), X_sim.end(), X_aug.begin());
        std::copy(X_sim.begin(), X_sim.end(), X_aug.begin() + NX_SINGLE);

        // Impone lo stato iniziale del problema al nodo 0.
        // ocp_nlp_constraints_model_set ha 7 argomenti; ocp_nlp_out_set ne ha 6.
        ocp_nlp_constraints_model_set(capsule->nlp_config, capsule->nlp_dims,
            capsule->nlp_in, capsule->nlp_out, 0, "lbx", X_aug.data());
        ocp_nlp_constraints_model_set(capsule->nlp_config, capsule->nlp_dims,
            capsule->nlp_in, capsule->nlp_out, 0, "ubx", X_aug.data());

        // Il nodo terminale ha parametri propri: aggiornare 0..N, non solo 0..N-1.
        const auto gait_horizon = gait_planner.get_gait_horizon(step, N + 1);
        if (gait_horizon.size() < static_cast<std::size_t>(N + 1)) {
            std::cerr << "Orizzonte del gait planner troppo corto allo step " << step << ".\n";
            break;
        }
        for (int k = 0; k <= N; ++k) {
            setStageParameters(capsule, k, gait_horizon[k], gait_planner, obstacle);
        }

        State terminal_guess = X_sol_t_prec[N];

        for (int k = 0; k < N; ++k) {
            const auto& gait_k = gait_horizon[k];
            const int idx_shift_u = (k < N - 1) ? k + 1 : k;
            const int idx_shift_x = k + 1;
            const State& x_prev = X_sol_t_prec[idx_shift_x];
            const Control& u_prev = U_sol_t_prec[idx_shift_u];

            const std::array<double, 3> state_op{x_prev[0], x_prev[1], x_prev[2]};
            const std::array<double, 3> state_pe{x_prev[11], x_prev[12], x_prev[13]};
            const auto hips_op = gait_planner.compute_hip_positions(state_op, gait_k);
            const auto hips_pe = gait_planner.compute_hip_positions(state_pe, gait_k);

            const auto closest_op = closestPoint({state_op[0], state_op[1]}, hips_op, obstacle);
            const auto closest_pe = closestPoint({state_pe[0], state_pe[1]}, hips_pe, obstacle);
            const double r_base = Config::r_obs + 0.1;
            const double margin_op = 1.8 * r_base;
            // Il modello pessimista aumenta il raggio con il tempo accumulato:
            // r_dynamic = p[6] + p[7] * x_pe[10] (stato complessivo indice 21).
            const double r_pe_guess = r_base + Config::y_dot_max * x_prev[21];
            const double margin_pe = 1.8 * r_pe_guess;
            const auto line_op = separatingLine(closest_op, obstacle, margin_op);
            const auto line_pe = separatingLine(closest_pe, obstacle, margin_pe);

            Control u_guess{};
            fillControlHalf(u_guess.data(), hips_op, u_prev[7], line_op);
            fillControlHalf(u_guess.data() + 11, hips_pe, u_prev[18], line_pe);

            // Come nel Python, il primo controllo optimistic coincide con quello pessimistic.
            if (k == 0) {
                std::copy(u_guess.begin() + 11, u_guess.end(), u_guess.begin());
            }
            ocp_nlp_out_set(capsule->nlp_config, capsule->nlp_dims,
                capsule->nlp_out, capsule->nlp_in, k, "u", u_guess.data());

            // Python non sovrascrive x al nodo 0: lo stato è già fissato da lbx/ubx.
            if (k > 0) {
                State x_guess{};
                // Ramo optimistic
                x_guess[0] = state_op[0];
                x_guess[1] = state_op[1];
                x_guess[2] = state_op[2];
                x_guess[3] = x_prev[3];
                x_guess[4] = x_prev[4];
                x_guess[5] = x_prev[5];
                x_guess[6] = x_prev[6];
                x_guess[7] = x_prev[7];
                x_guess[8] = x_prev[8];
                x_guess[9] = x_prev[9];
                x_guess[10] = u_prev[7];

                // Ramo pessimistic: gli appoggi sono gli elementi 17..20 di x_prev.
                x_guess[11] = state_pe[0];
                x_guess[12] = state_pe[1];
                x_guess[13] = state_pe[2];
                x_guess[14] = x_prev[14];
                x_guess[15] = x_prev[15];
                x_guess[16] = x_prev[16];
                x_guess[17] = x_prev[17];
                x_guess[18] = x_prev[18];
                x_guess[19] = x_prev[19];
                x_guess[20] = x_prev[20];
                x_guess[21] = u_prev[18];

                ocp_nlp_out_set(capsule->nlp_config, capsule->nlp_dims,
                    capsule->nlp_out, capsule->nlp_in, k, "x", x_guess.data());
                terminal_guess = x_guess;
            }
        }

        // Python usa l'ultimo x_guess anche come guess del nodo terminale.
        ocp_nlp_out_set(capsule->nlp_config, capsule->nlp_dims,
            capsule->nlp_out, capsule->nlp_in, N, "x", terminal_guess.data());

        const int status = lipm_opti_pessi_acados_solve(capsule);
        if (status != ACADOS_SUCCESS) {
            std::cerr << "\nSolver Acados fallito allo step " << step
                      << " (status " << status << ").\n";
            break;
        }

        Control u_opt{};
        State x_next{};
        ocp_nlp_out_get(capsule->nlp_config, capsule->nlp_dims,
            capsule->nlp_out, 0, "u", u_opt.data());
        ocp_nlp_out_get(capsule->nlp_config, capsule->nlp_dims,
            capsule->nlp_out, 1, "x", x_next.data());

        // Aggiorna tutto il warm start prima di sovrascrivere lo stato misurato.
        for (int k = 0; k < N; ++k) {
            ocp_nlp_out_get(capsule->nlp_config, capsule->nlp_dims,
                capsule->nlp_out, k, "x", X_sol_t_prec[k].data());
            ocp_nlp_out_get(capsule->nlp_config, capsule->nlp_dims,
                capsule->nlp_out, k, "u", U_sol_t_prec[k].data());
        }
        ocp_nlp_out_get(capsule->nlp_config, capsule->nlp_dims,
            capsule->nlp_out, N, "x", X_sol_t_prec[N].data());

        // Il Python applica al robot il ramo pessimistic (indici 11..21).
        std::copy(x_next.begin() + NX_SINGLE, x_next.end(), X_sim.begin());
        dt_chosen = u_opt[7];
        t_global += dt_chosen;

        log_file << step << ',' << X_sim[0] << ',' << X_sim[1] << ','
                 << X_sim[2] << ',' << X_sim[3] << ',' << X_sim[4] << '\n';
        std::cout << "Step " << step << " | Pos: [" << X_sim[0] << ", "
                  << X_sim[1] << "] | Theta: " << X_sim[2]
                  << " | dt: " << dt_chosen * 1000.0 << " ms\n";

        const double dist_target = norm2D({
            X_sim[0] - Config::c_target[0],
            X_sim[1] - Config::c_target[1]});
        if (dist_target < 0.1) {
            ++reached_target_flag;
            if (reached_target_flag >= 5) {
                std::cout << "Target Reached!\n";
                break;
            }
        } else {
            reached_target_flag = 0;
        }
    }

    log_file.close();
    lipm_opti_pessi_acados_free(capsule);
    lipm_opti_pessi_acados_free_capsule(capsule);
    return 0;
}
